A1: Window: requestAnimationFrame() method. This code starts a 
 step function that tracks time and updates the element position 
 or canvas frame on every screen refresh. 

JS(same as ref 28.1);
const element = document.getElementById("some-element-you-want-to-animate");
let start = null;

function step(timestamp) {
  if (start === null) start = timestamp;
  const progress = timestamp - start;
  
  // Move element right by up to 200px
  element.style.transform = `translateX(${Math.min(progress / 10, 200)}px)`;

  if (progress < 2000) {
    requestAnimationFrame(step);
  }
}
requestAnimationFrame(step);

JavaScript アニメーションコードjavascriptconst canvas = document.getElementById("myCanvas");
const ctx = canvas.getContext("2d");

// ボールの初期状態
let x = 50;
let y = 50;
let dx = 2; // X軸の移動速度
let dy = 3; // Y軸の移動速度
const radius = 15;

function animate() {
  // 1. 前のフレームの描画をクリア
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  // 2. ボールを描画
  ctx.beginPath();
  ctx.arc(x, y, radius, 0, Math.PI * 2);
  ctx.fillStyle = "#0095DD";
  ctx.fill();
  ctx.closePath();

  // 3. 壁との衝突判定（跳ね返り）
  if (x + dx > canvas.width - radius || x + dx < radius) {
    dx = -dx;
  }
  if (y + dy > canvas.height - radius || y + dy < radius) {
    dy = -dy;
  }

  // 4. 座標を更新
  x += dx;
  y += dy;

  // 5. 次のフレームをリクエスト
  requestAnimationFrame(animate);
}

// アニメーションを開始
requestAnimationFrame(animate);
コードは注意してご使用ください。

getContext("2d"): 
  Canvasに2Dグラフィックスを描画するためのコンテキストを取得します。

  毎フレーム必ず全画面を消去します。
